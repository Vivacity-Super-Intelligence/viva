cmake_minimum_required(VERSION 3.20)
project(viva VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VIVA_SANITIZE "Enable host address and undefined behavior sanitizers" OFF)
if(MSVC)
  add_compile_options(/W4 /WX)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Werror)
  if(VIVA_SANITIZE)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
  endif()
endif()
add_library(viva_core firmware/core/src/viva_core.c firmware/core/src/viva_protocol.c firmware/tests/viva_selftest.c)
target_include_directories(viva_core PUBLIC firmware/core/include firmware/tests)
enable_testing()
add_executable(viva_core_tests firmware/core/tests/test_core.c)
target_link_libraries(viva_core_tests PRIVATE viva_core)
add_test(NAME viva_core COMMAND viva_core_tests)
add_executable(viva_protocol_tests tests/unit/test_protocol.c)
target_link_libraries(viva_protocol_tests PRIVATE viva_core)
add_test(NAME viva_protocol COMMAND viva_protocol_tests)
add_executable(viva_host tests/integration/viva_host.c)
target_link_libraries(viva_host PRIVATE viva_core)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME viva_profiles COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/validation/validate_profiles.py)
add_test(NAME viva_host_tests COMMAND ${Python3_EXECUTABLE} -m unittest discover -s ${CMAKE_SOURCE_DIR}/tests/integration -p test_*.py)
set_tests_properties(viva_host_tests PROPERTIES ENVIRONMENT "VIVA_HOST=$<TARGET_FILE:viva_host>")
